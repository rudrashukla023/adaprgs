#include<stdio.h>
#include<stdlib.h>

int x[10];

int place(int x[], int k)
{
    int i;

    for(i=1; i<k; i++)
        if(x[i]==x[k] ||
           i-x[i]==k-x[k] ||
           i+x[i]==k+x[k])
            return 0;

    return 1;
}

void placequeens(int k, int x[], int n)
{
    while(k > 0)
    {
        x[k] = x[k] + 1;

        while((x[k] <= n) && (!place(x,k)))
            x[k] = x[k] + 1;

        if(x[k] <= n)
        {
            if(k == n)
                return;
            else
            {
                k++;
                x[k] = 0;
            }
        }
        else
            k--;
    }
}

int main()
{
    int q = 1, n, i;

    x[q] = 0;

    printf("Enter the number of queens: ");
    scanf("%d", &n);

    placequeens(q, x, n);

    for(i=1; i<=n; i++)
        printf("\nThe queen %d is placed at (%d, %d)", i, i, x[i]);

    return 0;
}
