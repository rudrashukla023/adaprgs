#include <stdio.h>
#define INF 999

void Dijkstra(int cost[10][10], int n, int start, int d[10]) {
    int visited[10] = {0}, i, j, min, u;
    for (i = 0; i < n; i++)
         d[i] = cost[start][i];
    d[start] = 0;
    visited[start] = 1;
    for (i = 1; i < n; i++) {
        min = INF; u = -1;
        for (j = 0; j < n; j++)
            if (!visited[j] && d[j] < min) { min = d[j]; u = j; }
        if (u == -1) break;
        visited[u] = 1;
        for (j = 0; j < n; j++)
            if (!visited[j] && d[u] + cost[u][j] < d[j])
                d[j] = d[u] + cost[u][j];
    }
}

int main() {
    int cost[10][10], d[10], n, start, i, j;
    scanf("%d", &n);
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++) {
            scanf("%d", &cost[i][j]);
            if (i != j && cost[i][j] == 0) cost[i][j] = INF;
        }
    scanf("%d", &start);
    Dijkstra(cost, n, start, d);
    for (i = 0; i < n; i++) printf("%d ", d[i]);
    return 0;
}
