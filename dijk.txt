##include <stdio.h>
#define INF 999

void Dijkstra(int cost[10][10], int n, int start, int d[10]) {
    int visited[10] = {0}, i, j, min, u;
 
    for (i = 0; i < n; i++)
        d[i] = cost[start][i];
    d[start] = 0;
    visited[start] = 1;

    for (i = 1; i < n; i++) {
        min = INF;
        u = -1;

        for (j = 0; j < n; j++) {
            if (!visited[j] && d[j] < min) {
                min = d[j];
                u = j;
            }
        }

        if (u == -1) break; // No reachable vertex left

        visited[u] = 1;

        for (j = 0; j < n; j++) {
            if (!visited[j] && d[u] + cost[u][j] < d[j]) {
                d[j] = d[u] + cost[u][j];
            }
        }
    }
}

int main() {
    int cost[10][10], d[10], n, start, i, j;

    printf("Enter the number of vertices: ");
    scanf("%d", &n);

    printf("Enter the cost matrix (use 0 for no edge):\n");
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) {
            scanf("%d", &cost[i][j]);
            // Convert non-diagonal 0s into INF
            if (i != j && cost[i][j] == 0) {
                cost[i][j] = INF;
            }
        }
    }

    printf("Enter the starting vertex (0 to %d): ", n - 1);
    scanf("%d", &start);

    Dijkstra(cost, n, start, d);

    printf("The shortest distances from vertex %d are:\n", start);
    for (i = 0; i < n; i++)
        printf("To vertex %d: %d\n", i, d[i]);

    return 0;
}
