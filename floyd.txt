#include <stdio.h>
#include <stdlib.h>
#define INF 999

int main() {
    int n, u, v, k;
    printf("N: ");
    scanf("%d", &n);

    int P[n][n];
    printf("Input Adjacency Matrix:-\n");
    for (u = 0; u < n; ++u) {
        for (v = 0; v < n; ++v) {
            scanf("%d", &P[u][v]);
            if (!P[u][v] && u != v)   // replace 0 with INF for no edge
                P[u][v] = INF;
        }
    }

    
    for (k = 0; k < n; ++k)
        for (u = 0; u < n; ++u)
            for (v = 0; v < n; ++v)
                if (P[u][v] > P[u][k] + P[k][v])
                    P[u][v] = P[u][k] + P[k][v];

    printf("\nShortest Path Pairs:-\n");
    for (u = 0; u < n; ++u) {
        for (v = 0; v < n; ++v) {
            if (P[u][v] == INF)
                printf("INF ");
            else
                printf("%d ", P[u][v]);
        }
        printf("\n");
    }

    return 0;
}
